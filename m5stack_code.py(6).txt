def rgb_set(r, g, b):
    np[0] = (r, g, b)
    np.write()

def rgb_off():
    rgb_set(0, 0, 0)

def rgb_red():
    rgb_set(255, 0, 0)

# ================= ENV SENSOR =================
i2c = I2C(0, scl=Pin(22), sda=Pin(21), freq=100000)
DHT12_ADDR = 0x5C

def read_dht12():
    data = i2c.readfrom_mem(DHT12_ADDR, 0x00, 5)
    hum = data[0] + data[1] / 10.0
    temp = data[2] + data[3] / 10.0
    chk = (data[0] + data[1] + data[2] + data[3]) & 0xFF
    if chk != data[4]:
        raise Exception("DHT12 checksum error")
    return float(temp), float(hum)

def safe_read_env():
    try:
        return read_dht12()
    except Exception as e:
        print("ENV error:", e)
        return None, None

ENV_PUBLISH_EVERY_MS = 2000
last_env_pub_ms = 0

def publish_env_if_due(now_ms):
    global last_env_pub_ms

    if utime.ticks_diff(now_ms, last_env_pub_ms) < ENV_PUBLISH_EVERY_MS:
        return

    last_env_pub_ms = now_ms
    t, h = safe_read_env()

    if t is None:
        tempLabel.setText("Room Temp: --.- C")
    else:
        tempLabel.setText("Room Temp: %.1f C" % t)

    if h is None:
        humLabel.setText("Cellar Humidity: --.- %")
    else:
        humLabel.setText("Cellar Humidity: %.1f %%" % h)

    payload = {
        "room": "living",
        "ts": int(now_ms)
    }

    if t is not None:
        payload["tempC"] = round(t, 2)

    if h is not None:
        payload["hum"] = round(h, 1)

    try:
        client.publish(MQTT_ENV_LIVING, ujson.dumps(payload).encode())

        if PUBLISH_TO_KELLER:
            payload_k = dict(payload)
            payload_k["room"] = "cellar"
            client.publish(MQTT_ENV_KELLER, ujson.dumps(payload_k).encode())

        print("ENV published:", payload)

    except Exception as e:
        print("ENV publish error:", e)

# ================= STATUS HELPERS =================
def format_status_message(text):
    t = text.strip()
    t = t.replace("?", "")
    t = t.replace("turned ", "")
    t = t.replace("  ", " ")
    return t

def is_relevant_device(text_lower):
    return ("window" in text_lower) or ("door" in text_lower) or ("lock" in text_lower)

def set_rgb_from_message(text):
    t = text.lower()

    if not is_relevant_device(t):
        return

    if "unlocked" in t:
        rgb_green()
        return

    if "locked" in t:
        rgb_red()
        return

    if "open" in t or "off" in t:
        rgb_green()
        return

    if "closed" in t or "on" in t:
        rgb_red()
        return

# ================= MQTT CALLBACK =================
def mqtt_callback(topic, msg):
    try:
        decoded = msg.decode().strip()

        if decoded.startswith("{"):
            print("Ignored JSON:", decoded)
            return

        if decoded.lower() == "alive 1" or decoded.lower().startswith("alive"):
            print("Ignored alive message:", decoded)
            return

        pretty = format_status_message(decoded)

        msgLabel.setText(pretty)
        print("MQTT message:", pretty)

        set_rgb_from_message(pretty)

    except Exception as e:
        print("MQTT callback error:", e)

# ================= MQTT HELPERS =================
def publish_motion(payload_str):
    try:
        client.publish(MQTT_MOTION_TOPIC, payload_str.encode())
        print("MOTION published:", payload_str)
    except Exception as e:
        print("Publish error:", e)

def turn_off_other_room(new_mode):
    if new_mode == MODE_CELLAR:
        publish_motion("STORE_OFF")
    elif new_mode == MODE_STORE_ROOM:
        publish_motion("KELLER_OFF")
    elif new_mode == MODE_OFF:
        publish_motion("KELLER_OFF")
        publish_motion("STORE_OFF")

def set_mode(new_mode):
    global mode, auto_on, last_motion_ms, LAST_KEEPALIVE_MS

    mode = new_mode
    auto_on = False
    last_motion_ms = 0

    update_mode_label()
    msgLabel.setText("Mode changed")

    utime.sleep_ms(200)
    rgb_off()
